  static const Color aiCyan = headerMint;

  /// Bleu vif — CTA, FAB, item actif.

  static const Color darkSurface = Color(0xFF0C1C2A);
  static const Color darkCard = Color(0xFF122636);
  static const Color darkBorder = Color(0xFF1E3A52);
  static const Color darkText = Color(0xFFF0F7FC);
  static const Color darkTextSecondary = Color(0xFFB8C0CC);

  // ——— Verre iOS (nav / chrome flottant) ———
  static Color glassLight = const Color(0xFFF8FBFE).withValues(alpha: 0.78);
  static Color glassDark = const Color(0xFF0C1C2A).withValues(alpha: 0.78);
  static Color glassBorder = Colors.white.withValues(alpha: 0.55);

  /// Surfaces adaptées light / dark (composants partagés).
  static bool isDark(BuildContext context) =>
      Theme.of(context).brightness == Brightness.dark;

  static Color card(BuildContext context) =>
      isDark(context) ? darkCard : surfaceBright;

  static Color elevated(BuildContext context) =>
      isDark(context) ? darkSurface : surfaceContainerLowest;

  static Color mutedFill(BuildContext context) =>
      isDark(context) ? darkSurface : primaryFixed;

  static Color glass(BuildContext context, {bool performance = false}) {
    if (isDark(context)) {
      return darkCard.withValues(alpha: performance ? 0.96 : 0.72);
    }
    return surfaceBright.withValues(alpha: performance ? 0.96 : 0.78);
  }

  static Color glassStroke(BuildContext context) => isDark(context)
      ? Colors.white.withValues(alpha: 0.10)
      : Colors.white.withValues(alpha: 0.72);

  /// Ombre teintée ciel — jamais noir pur sur fond clair.
  static List<BoxShadow> cardShadow({
    double opacity = 0.08,
    bool dark = false,
  }) => [
    BoxShadow(
      color: (dark ? Colors.black : const Color(0xFF1A7BC7)).withValues(
        alpha: dark ? opacity * 2.2 : opacity,
      ),
      blurRadius: 24,
      offset: const Offset(0, 12),
      spreadRadius: -8,
    ),
  ];

  /// Ombre douce turquoise pour accents positifs.
  static List<BoxShadow> accentShadow({double opacity = 0.18}) => [
    BoxShadow(
      color: turquoise.withValues(alpha: opacity),
      blurRadius: 20,
      offset: const Offset(0, 10),
      spreadRadius: -6,
    ),
  ];

    begin: Alignment.topLeft,

  static const LinearGradient brandGradient = LinearGradient(
    begin: Alignment.topLeft,

  );

    BoxShadow(
      color: color.withValues(alpha: opacity),
      blurRadius: 20,
      offset: const Offset(0, 10),
      spreadRadius: -8,
    ),
  ];
